local function setStatus(t)
	if statusLbl then statusLbl.Text = t end
end

local function hrp()
	local c = LP.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function isDay()
	if not AreaEggCycle then return true end
	local ok, night = pcall(function()
		return AreaEggCycle.IsNightPhase(workspace:GetServerTimeNow())
	end)
	return not (ok and night == true)
end

local function wallOpen()
	local a = workspace:FindFirstChild("__OBJECTS")
	a = a and a:FindFirstChild("Areas")
	if not a then return true end
	local v = a:FindFirstChild("WallStartVisual")
	if v and v:IsA("BasePart") and v.Size.Y > 0.25 then return false end
	local c = a:FindFirstChild("WallStartCollision")
	if c and c:IsA("BasePart") and c.CanCollide then return false end
	return true
end

local function canSteal()
	if not S.WaitDay then return true end
	return isDay() and wallOpen()
end

local function rarityOf(rec)
	if not Assets or not Assets.Directory then return "" end
	local e = Assets.Directory[rec.AssetCategory]
	if not e or not e.Rarity then return "" end
	return e.Rarity.DisplayName or e.Rarity.Name or ""
end

local function valueOf(rec)
	if not Assets or not Assets.Directory then return 0 end
	local e = Assets.Directory[rec.AssetCategory]
	if not e then return 0 end
	local rate = tonumber(e.EarningRate) or 0
	local scale = tonumber(rec.AssetScale) or 1
	local mult = 1
	if rec.Mutations and Mutations and type(Mutations.EarningsFor) == "function" then
		local ok, r = pcall(Mutations.EarningsFor, rec.Mutations)
		if ok then mult = tonumber(r) or 1 end
	end
	return math.floor(rate * scale * mult + 0.5)
end

local function isStartEgg(rec)
	if SlotId and type(SlotId.LooksLikeFirstAreaUid) == "function" then
		local ok, r = pcall(SlotId.LooksLikeFirstAreaUid, rec.Uid)
		if ok and r then return true end
	end
	if rec.BottomCFrame then
		return (rec.BottomCFrame.Position - START).Magnitude < 100
	end
	return false
end

local function snapshot()
	if not AskSnap then return {} end
	local ok, res = pcall(function() return AskSnap:InvokeServer() end)
	if not ok or type(res) ~= "table" or not res.Records then return {} end
	return res.Records
end

local function fetchEggs()
	local records = snapshot()
	local startE, best = nil, {}
	for _, rec in ipairs(records) do
		if (rec.State == "Slot" or rec.State == "Dropped") and rec.BottomCFrame then
			local rar = rarityOf(rec)
			local data = {
				Uid = rec.Uid,
				NestId = rec.NestId,
				AreaId = rec.AreaId,
				Rarity = rar,
				Name = (Assets and Assets.Directory and Assets.Directory[rec.AssetCategory] and Assets.Directory[rec.AssetCategory].DisplayName) or "?",
				Value = valueOf(rec),
				Position = rec.BottomCFrame.Position,
				Record = rec,
			}
			if isStartEgg(rec) and not startE then
				startE = data
			end
			local key = string.lower(tostring(rar)):gsub("%s+", "")
			local any = false
			for _, on in pairs(Rarities) do
				if on then any = true break end
			end
			if (not any or Rarities[key]) and (not startE or data.Uid ~= startE.Uid) then
				table.insert(best, data)
			end
		end
	end
	table.sort(best, function(a, b) return a.Value > b.Value end)
	return best, startE
end

-- Lennon-style carry (EggState.CarryFieldEgg preferred)
local function carryEgg(egg)
	if not egg or not egg.Uid then return false end
	if EggState and type(EggState.CarryFieldEgg) == "function" then
		local nestKey = nil
		if SlotId and egg.NestId and type(SlotId.LooksLikeFirstAreaUid) == "function" then
			local ok, isFirst = pcall(SlotId.LooksLikeFirstAreaUid, egg.Uid)
			if ok and isFirst and type(SlotId.SlotKey) == "function" then
				pcall(function()
					nestKey = SlotId.SlotKey(egg.AreaId, egg.NestId)
				end)
			end
		end
		local ok, r = pcall(function()
			return EggState.CarryFieldEgg(egg.Uid, nestKey)
		end)
		if ok and r then return true end
	end
	if AskCarry then
		local ok, r = pcall(function()
			return AskCarry:InvokeServer({ Uid = egg.Uid })
		end)
		return ok and r == true
	end
	return false
end

local function spamCarry(egg, times)
	times = times or 25
	for i = 1, times do
		if not S.Enabled then return false end
		if carryEgg(egg) then return true end
		task.wait(0.04)
	end
	return false
end

local function tweenTo(pos, speed)
	local h = hrp()
	if not h or not pos then return false end
	speed = speed or S.Speed
	local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.PlatformStand = true
	end
	h.AssemblyLinearVelocity = Vector3.zero
	h.AssemblyAngularVelocity = Vector3.zero
	local target = pos + Vector3.new(0, 3, 0)
	local dist = (h.Position - target).Magnitude
	if dist < 4 then
		if hum then hum.PlatformStand = false end
		return true
	end
	local tw = TweenService:Create(
		h,
		TweenInfo.new(math.max(dist / math.max(speed, 50), 0.05), Enum.EasingStyle.Linear),
		{ CFrame = CFrame.new(target) }
	)
	local conn
	conn = RunService.Heartbeat:Connect(function()
		if h and h.Parent then
			h.AssemblyLinearVelocity = Vector3.zero
			h.AssemblyAngularVelocity = Vector3.zero
		end
	end)
	tw:Play()
	tw.Completed:Wait()
	if conn then conn:Disconnect() end
	if h and h.Parent then
		h.AssemblyLinearVelocity = Vector3.zero
	end
	if hum then hum.PlatformStand = false end
	return true
end

-- Wait for CarryChanged (Lennon style)
local function waitCarry(uid, timeout)
	timeout = timeout or 8
	if not EggState or not EggState.CarryChanged then
		task.wait(0.5)
		return true
	end
	local got = false
	local conn
	conn = EggState.CarryChanged:Connect(function(info)
		if info and info.IsCarrying and info.Uid == uid then
			got = true
		end
	end)
	local t0 = tick()
	while not got and tick() - t0 < timeout do
		if not S.Enabled then break end
		task.wait(0.05)
	end
	if conn then conn:Disconnect() end
	return got
end

-- ===== TELEGUIADO CYCLE (Lennon) =====
local function runTeleguiado()
	if busy or not S.Enabled then return end
	if not canSteal() then
		setStatus(isDay() and "Gate..." or "Dia...")
		return
	end
	busy = true
	setStatus("Inicio...")

	-- 1) Ir al punto inicio
	tweenTo(START, S.Speed)
	if not S.Enabled then busy = false return end

	local best, startE = fetchEggs()

	-- 2) Robar huevo inicio (obligatorio estilo Lennon)
	if startE then
		setStatus("Robando inicio")
		tweenTo(startE.Position, S.Speed)
		task.spawn(function()
			spamCarry(startE, 20)
		end)
		local ok = waitCarry(startE.Uid, 6)
		if not ok then
			-- reintentar spam
			spamCarry(startE, 15)
			waitCarry(startE.Uid, 3)
		end
	end

	if not S.Enabled then busy = false setStatus("Off") return end

	-- 3) Ir al mejor huevo filtrado
	best = select(1, fetchEggs())
	local target = best and best[1]
	if not target then
		setStatus("Sin rareza")
		if S.GoSafe then tweenTo(SAFE, S.Speed) end
		busy = false
		return
	end

	setStatus("-> " .. (target.Rarity or "?") .. " " .. (target.Name or ""))
	tweenTo(target.Position, S.Speed)
	if not S.Enabled then busy = false return end

	setStatus("Robando " .. (target.Rarity or ""))
	task.spawn(function()
		spamCarry(target, 40)
	end)
	local grabbed = waitCarry(target.Uid, 8)
	if not grabbed then
		spamCarry(target, 20)
	end

	-- 4) Volver a safe + proteger
	if S.GoSafe then
		setStatus("Safe zone")
		task.spawn(function()
			for i = 1, 20 do
				if not S.Enabled then break end
				carryEgg(target)
				task.wait(0.06)
			end
		end)
		tweenTo(SAFE, math.min(S.Speed, 600))
	end

	setStatus(S.Loop and "Listo (loop)" or "Listo")
	busy = false
end
